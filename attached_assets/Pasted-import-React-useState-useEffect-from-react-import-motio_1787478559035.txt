export default function App() {
  const [selectedConversationId, setSelectedConversationId] = useState<string>('7');
  const [conversations, setConversations] = useState<Conversation[]>([]);
  const [selectedTransaction, setSelectedTransaction] = useState<Transaction | null>(null);
  const [loading, setLoading] = useState(true);

  // Seeding Logic
  useEffect(() => {
    async function seedData() {
      const batch = writeBatch(db);
      SEED_CONVERSATIONS.forEach(c => {
        batch.set(doc(db, 'conversations', c.id), c);
      });
      SEED_TRANSACTIONS.forEach(t => {
        batch.set(doc(db, 'transactions', t.id), t);
      });
      await batch.commit();
    }
    seedData();
  }, []);

  // Real-time Subscriptions
  useEffect(() => {
    const unsub = subscribeToConversations((data) => {
      setConversations(data);
      setLoading(false);
    });
    return () => unsub();
  }, []);

  useEffect(() => {
    if (selectedConversationId) {
      const unsub = subscribeToTransaction(selectedConversationId, (data) => {
        setSelectedTransaction(data);
      });
      return () => unsub();
    }
  }, [selectedConversationId]);

  if (loading && conversations.length === 0) {
    return (
      <div className="h-screen w-screen flex items-center justify-center bg-slate-50">
        <div className="flex flex-col items-center gap-6">
          <motion.div 
            animate={{ rotate: 360 }}
            transition={{ repeat: Infinity, duration: 1.5, ease: "linear" }}
            className="w-12 h-12 border-[3px] border-blue-500/20 border-t-blue-600 rounded-full shadow-lg shadow-blue-100" 
          />
          <div className="flex flex-col items-center gap-1">
            <p className="text-slate-800 font-black text-sm tracking-tight">LEO Panel</p>
            <p className="text-slate-400 text-[10px] font-bold uppercase tracking-widest animate-pulse">Initializing Interface...</p>
          </div>
        </div>
      </div>
    );
  }

  return (
    <div className="flex flex-col h-screen w-screen overflow-hidden bg-white text-gray-900 font-sans selection:bg-blue-100">
      <TopBar title="صندوق الوارد" />
      
      <div className="flex flex-1 overflow-hidden" dir="ltr">
        <LeftNav />
        
        <div className="flex flex-1 overflow-hidden" dir="rtl">
          <Sidebar 
            conversations={conversations} 
            selectedId={selectedConversationId}
            onSelect={setSelectedConversationId}
          />
          
          <MainContent 
            transaction={selectedTransaction} 
          />
        </div>
      </div>
    </div>
  );
}
